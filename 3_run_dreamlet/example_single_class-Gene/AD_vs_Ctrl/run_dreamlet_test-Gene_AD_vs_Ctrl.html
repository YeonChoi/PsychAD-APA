form = ~ 0 + DxAD + sex + scale(age) + scale(pmi) + log(n_genes) + percent_mito + mito_ribo + ribo_genes + (1|individualID)
#contrasts = c(AD_vs_Ctrl = "DxADAD - DxADCtrl")

# If individualID level, remove individualID
if (PSB_SAMPLE_LEVEL == 'individualID'){
  form = update(form,  ~ . - (1|individualID))
}

# If multiple brain bank
if (length(unique(colData(res.proc)$Source)) > 1){
  form = update(form,  ~ . + (1|Source))
}
